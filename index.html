let slides = document.querySelectorAll(".hero-slide"); let dots = document.querySelectorAll(".dot"); let index = 0; let interval; /* afficher slide */ function showSlide(i){ slides.forEach(s => s.classList.remove("active")); dots.forEach(d => d.classList.remove("active")); slides[i].classList.add("active"); dots[i].classList.add("active"); index = i; } /* auto */ function startAuto(){ interval = setInterval(() => { index++; if(index >= slides.length){ index = 0; } showSlide(index); }, 8000); } /* pause au survol */ let hero = document.querySelector(".hero-expert"); hero.addEventListener("mouseenter", () => clearInterval(interval)); hero.addEventListener("mouseleave", startAuto); /* dots clic */ dots.forEach((dot, i) => { dot.addEventListener("click", () => showSlide(i)); }); /* lancer */ startAuto();
